for (let i = ids.length - 1; i >= 0; i--) { if (String(ids[i][0]) === String(rowObj.sessionId)) { targetRow = i + 2; break; } } } const rowValues = [ rowObj.sessionId, rowObj.nowStr, rowObj.igAccount, rowObj.lastStep, rowObj.s1, rowObj.s2, rowObj.s3, rowObj.s4, rowObj.s5, rowObj.totalSec, rowObj.specNote ]; if (!rowObj.specNote && existingNote) rowValues[10] = existingNote; sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]); } else { sheet.appendRow(rowValues); } } function sendOrderEmail(data) { const recipient = OWNER_EMAIL || Session.getActiveUser().getEmail();
